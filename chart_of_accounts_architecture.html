جدول الحسابات الرئيسي — GL_ACCOUNT

GL_ACCOUNT (Tenant Schema)
🔑 ID NUMBER(19) GENERATED BY DEFAULT AS IDENTITY ACCOUNT_CODE VARCHAR2(20) NOT NULL UNIQUE -- '1-01-01-02-0001' ACCOUNT_NAME_AR NVARCHAR2(200) NOT NULL ACCOUNT_NAME_EN VARCHAR2(200) NOT NULL 🔗 CATEGORY_ID NUMBER(19) NOT NULL → ACCOUNT_CATEGORY.ID 🔗 SUB_CATEGORY_ID NUMBER(19) NULL → ACCOUNT_SUB_CATEGORY.ID 🔗 PARENT_ACCOUNT_ID NUMBER(19) NULL → GL_ACCOUNT.ID (Self-ref) ACCOUNT_TYPE VARCHAR2(10) NOT NULL -- 'HEADER' or 'DETAIL' NORMAL_BALANCE CHAR(1) NOT NULL -- 'D' or 'C' ACCOUNT_LEVEL NUMBER(1) NOT NULL -- 1, 2, 3, 4, or 5 IS_CONTRA NUMBER(1) DEFAULT 0 IS_CONTROL_ACCOUNT VARCHAR2(10) NULL -- 'AR', 'AP', or NULL IS_POSTABLE NUMBER(1) GENERATED ALWAYS AS (CASE WHEN ACCOUNT_TYPE='DETAIL' AND IS_ACTIVE=1 THEN 1 ELSE 0 END) VIRTUAL IS_ACTIVE NUMBER(1) DEFAULT 1 HAS_BRANCHES NUMBER(1) DEFAULT 0 -- تقييد بفروع معينة؟ DESCRIPTION_AR NVARCHAR2(500) DESCRIPTION_EN VARCHAR2(500) CREATION_USER VARCHAR2(100) NOT NULL CREATION_DATE TIMESTAMP DEFAULT SYSTIMESTAMP UPDATE_USER VARCHAR2(100) UPDATE_DATE TIMESTAMP

الحسابHAS_BRANCHESالسبب
1-01-01-01-0001 الصندوق1 = مقيدكل فرع له صندوقه المستقل
1-01-01-02-0001 البنك1 = مقيدحساب بنكي خاص بفرع

6-02-01-01-0001 رواتب1 = مقيدكل فرع يتابع مصاريفه

القاعدة

HAS_BRANCHES = 0 → الحساب متاح لجميع الفروع (الافتراضي). HAS_BRANCHES = 1 → فقط الفروع المسجلة في GL_ACCOUNT_BRANCH.

public class GlAccount
{
    public long Id { get; set; }

    public char NormalBalance { get; set; }     // 'D' / 'C'

    public bool IsContra { get; set; }

    public bool IsPostable { get; set; }       // Virtual column

}

القواعد الذهبية

✅ افعل

  • استخدم PARENT_ACCOUNT_ID حصرياً للهرمية
  • الترحيل فقط على المستوى 5 (DETAIL)
  • ازرع حسابات MVP الأربعة مع كل شركة جديدة
  • اختبر SUM(subledger) = control_account دورياً
  • سجّل كل تعديل في GL_ACCOUNT_AUDIT_LOG

🚫 لا تفعل

  • لا تستنتج العلاقة الأبوية من الرقم أبداً
  • لا تسمح بالترحيل على HEADER (L1-L4)
  • لا تسمح بالترحيل اليدوي على AR/AP
  • لا تحذف حساباً استُخدم في قيد
  • لا تنسخ الشجرة لكل فرع
